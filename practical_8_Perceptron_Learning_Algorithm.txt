# Practical 8: Perceptron Learning Algorithm
import numpy as np

# Input data
X = np.array([
    [2, 3],
    [3, 4],
    [4, 5],
    [1, 1],
    [2, 1],
    [1, 2]
])

# Target outputs
y = np.array([1, 1, 1, 0, 0, 0])

# Initialize weights and bias
weights = np.zeros(2)
bias = 0
learning_rate = 0.1

# Perceptron learning (stops early once an epoch has no errors)
for epoch in range(1, 101):
    total_errors = 0
    for i in range(len(X)):
        z = np.dot(X[i], weights) + bias
        prediction = 1 if z >= 0 else 0
        error = y[i] - prediction
        if error != 0:
            weights = weights + learning_rate * error * X[i]
            bias = bias + learning_rate * error
            total_errors += 1
    print(f"Epoch {epoch}: errors = {total_errors}")
    if total_errors == 0:
        break

# Display trained parameters
print("\nWeights:", weights)
print("Bias:", bias)

# Test the network
test_data = np.array([
    [5, 6],
    [1, 1]
])

print("\nTesting")
for point in test_data:
    z = np.dot(point, weights) + bias
    prediction = 1 if z >= 0 else 0
    print(point, "=> Class", prediction)
