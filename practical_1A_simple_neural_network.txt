# Practical 1A: Design a Simple Neural Network (SNN)
x = float(input("Enter the input x = "))
b = float(input("Enter the bias b = "))
w = float(input("Enter the weight w = "))

# Calculating net input value
net = (x * w) + b
print("Net =", net)

# Activation function (ramp)
if net < 0:
    out = 0
elif 0 <= net <= 1:
    out = net
else:
    out = 1

print("Output =", out)
