# Practical 1B: Binary and Bipolar Sigmoidal Functions
import math

x = float(input("Enter the input x = "))
b = float(input("Enter the bias b = "))
w = float(input("Enter the weight w = "))

# Net input value
net = (x * w) + b
print("Net =", net)

# Binary sigmoidal function
binary_sigmoidal = 1 / (1 + math.exp(-net))

# Bipolar sigmoidal function
bipolar_sigmoidal = (1 - math.exp(-net)) / (1 + math.exp(-net))

print("Binary Sigmoidal =", round(binary_sigmoidal, 2))
print("Bipolar Sigmoidal =", round(bipolar_sigmoidal, 4))
